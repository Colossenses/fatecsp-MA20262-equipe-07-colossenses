// CONSULTA 12: dCalendario
let
    DataInicial = #date(2023, 10, 1),
    DataFinal = #date(2027, 5, 1),
    Quantidade = Duration.Days(DataFinal - DataInicial) + 1,
    Datas = List.Dates(DataInicial, Quantidade, #duration(1,0,0,0)),
    Tabela = Table.FromList(Datas, Splitter.SplitByNothing(), {"Data"}),
    Tipo = Table.TransformColumnTypes(Tabela, {{"Data", type date}}),
    Ano = Table.AddColumn(Tipo, "Ano", each Date.Year([Data]), Int64.Type),
    Mes = Table.AddColumn(Ano, "Mes", each Date.Month([Data]), Int64.Type),
    NomeMes = Table.AddColumn(Mes, "NomeMes", each Date.MonthName([Data], "pt-BR"), type text),
    AnoMes = Table.AddColumn(NomeMes, "AnoMes", each Date.ToText([Data], "yyyy-MM"), type text),
    DiaSemana = Table.AddColumn(AnoMes, "DiaSemana", each Date.DayOfWeek([Data], Day.Monday)+1, Int64.Type),
    NomeDia = Table.AddColumn(DiaSemana, "NomeDia", each Date.DayOfWeekName([Data], "pt-BR"), type text)
in
    NomeDia
