    SUM ( 'sptrans findividualpass'[Seg] )
        + SUM ( 'sptrans findividualpass'[Ter] )
        + SUM ( 'sptrans findividualpass'[Qua] )
        + SUM ( 'sptrans findividualpass'[Qui] )
        + SUM ( 'sptrans findividualpass'[Sex] )
        + SUM ( 'sptrans findividualpass'[Sab] )
        + SUM ( 'sptrans findividualpass'[Dom] )
)
    * CALCULATE (
        MIN ( 'sptrans dfareattributes'[price] );
        TREATAS (
            VALUES ( 'sptrans ffarerules'[fare_id] );
            'sptrans dfareattributes'[fare_id]
        )
    )

                    'sptrans dfareattributes'[fare_id]
